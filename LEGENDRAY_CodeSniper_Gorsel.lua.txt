local function corner(parent, radius)
    local obj = Instance.new("UICorner")
    obj.CornerRadius = UDim.new(0, radius or 8)
    obj.Parent = parent
    return obj
end

local function stroke(parent, color, thickness, transparency)
    local obj = Instance.new("UIStroke")
    obj.Color = color or C.line
    obj.Thickness = thickness or 1
    obj.Transparency = transparency or 0
    obj.Parent = parent
    return obj
end

local function label(parent, text, size, color, font)
    local obj = Instance.new("TextLabel")
    obj.BackgroundTransparency = 1
    obj.Text = text
    obj.TextColor3 = color or C.white
    obj.TextSize = size or 14
    obj.Font = font or Enum.Font.Gotham
    obj.TextXAlignment = Enum.TextXAlignment.Left
    obj.Parent = parent
    return obj
end

local visualStates = {}
local controlNames = {
    "Manuel",
    "Listen",
    "Speed",
    "Words",
    "Auto Redeem",
    "Auto Type",
    "Spam",
}

local function makeVisualControl(name, order)
    local button = Instance.new("TextButton")
    button.Name = name:gsub("%s+", "") .. "VisualButton"
    button.LayoutOrder = order
    button.BackgroundColor3 = C.panel2
    button.BorderSizePixel = 0
    button.AutoButtonColor = false
    button.Text = name .. "  •  OFF"
    button.TextColor3 = C.muted
    button.TextSize = 11
    button.Font = Enum.Font.GothamSemibold
    button.Parent = controlPanel
    corner(button, 7)
    stroke(button, C.line, 1)

    visualStates[name] = false
    button.MouseButton1Click:Connect(function()
        visualStates[name] = not visualStates[name]
        local enabled = visualStates[name]
        button.Text = name .. (enabled and "  •  ON" or "  •  OFF")
        button.BackgroundColor3 = enabled
            and Color3.fromRGB(12, 43, 68)
            or C.panel2
        button.TextColor3 = enabled and C.bright or C.muted
    end)
end

for index, name in ipairs(controlNames) do
    makeVisualControl(name, index)
end

local pageNames = {"Code Sniper", "Feed", "Ayarlar", "Test", "Loglar"}
local pageDescriptions = {
    "Visual code workspace preview.",
    "Sample feed and announcement area.",
    "Appearance and interface preferences.",
    "Interface preview and sample status.",
    "Local visual event history.",
}

local selectedButton
local function showPage(name, description, button)
    pageTitle.Text = name
    pageDesc.Text = description
    if selectedButton then
        selectedButton.BackgroundColor3 = C.panel2
        selectedButton.TextColor3 = C.muted
        local oldBar = selectedButton:FindFirstChild("ActiveBar")
        if oldBar then oldBar:Destroy() end
    end
    selectedButton = button
    if button then
        button.BackgroundColor3 = Color3.fromRGB(12, 43, 68)
        button.TextColor3 = C.white
        local bar = Instance.new("Frame")
        bar.Name = "ActiveBar"
        bar.Size = UDim2.new(0, 3, 1, -12)
        bar.Position = UDim2.fromOffset(0, 6)
        bar.BackgroundColor3 = C.bright
        bar.BorderSizePixel = 0
        bar.Parent = button
        corner(bar, 2)
    end
end

for i, name in ipairs(pageNames) do
    local button = Instance.new("TextButton")
    button.Name = name:gsub("%s+", "") .. "Button"
    button.Position = UDim2.fromOffset(10, 43 + (i - 1) * 51)
    button.Size = UDim2.new(1, -20, 0, 42)
    button.BackgroundColor3 = C.panel2
    button.BorderSizePixel = 0
    button.Text = "   " .. name
    button.TextColor3 = C.muted
    button.TextSize = 12
    button.Font = Enum.Font.GothamSemibold
    button.TextXAlignment = Enum.TextXAlignment.Left
    button.AutoButtonColor = false
    button.Parent = sidebar
    corner(button, 7)

    button.MouseEnter:Connect(function()
        if selectedButton ~= button then
            TweenService:Create(button, TweenInfo.new(0.12), {
                BackgroundColor3 = Color3.fromRGB(25, 37, 57)
            }):Play()
        end
    end)
    button.MouseLeave:Connect(function()
        if selectedButton ~= button then
            TweenService:Create(button, TweenInfo.new(0.12), {
                BackgroundColor3 = C.panel2
            }):Play()
        end
    end)
    button.MouseButton1Click:Connect(function()
        showPage(name, pageDescriptions[i], button)
    end)

    if i == 1 then
        showPage(name, pageDescriptions[i], button)
    end
end

-- Sürüklenebilir üst başlık
local dragging = false
local dragStart, startPosition
top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

top.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPosition.X.Scale, startPosition.X.Offset + delta.X,
            startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
        )
    end
end)
